#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
Оркестратор для создания аудиокниги
Версия 8.0 - модульная архитектура с MVC
"""

import sys
import argparse
import time
from pathlib import Path

# Сначала добавляем путь
sys.path.insert(0, str(Path(__file__).parent))

# Теперь все импорты вместе
from core.debug_core import init_debug, get_debugger, debug_trace, debug_timing
from core.dependency_manager import get_dependency_manager
from core.config_manager import get_config_manager
from core.context import ExecutionContext, ProcessingOptions, ProcessingMode, AudioFormat
from core.pipeline import Pipeline, PipelineStage
from core.exceptions import AudiobookError
from services.text_preparation import TextPreparationService
from services.ssml_validation import SSMLValidationService
from services.audio_generation import AudioGenerationService
from models.ssml_chunk import SSMLChunk


class AudiobookOrchestrator:
    """
    Главный оркестратор создания аудиокниги.
    Координирует все сервисы и управляет процессом.
    """

    def __init__(self, debug_level: str = None):
        # Инициализируем отладчик
        if debug_level:
            self.debug = init_debug(debug_level)
        else:
            self.debug = get_debugger()

        self.debug.log("=" * 60, level="INFO")
        self.debug.log("🎙️ AUDIOBOOK CREATOR v8.0", level="INFO")
        self.debug.log("=" * 60, level="INFO")

        # Инициализируем менеджеры
        self.debug.log("Инициализация зависимостей...", level="INFO")
        self.deps = get_dependency_manager()

        self.debug.log("Загрузка конфигураций...", level="INFO")
        self.config = get_config_manager()

        # Контекст выполнения (будет создан при запуске)
        self.context: ExecutionContext = None

        # Сервисы (создаются при необходимости)
        self._text_service: TextPreparationService = None
        self._validation_service: SSMLValidationService = None
        self._audio_service: AudioGenerationService = None

        # Статистика
        self.start_time = None
        self.end_time = None

    @property
    def text_service(self) -> TextPreparationService:
        if self._text_service is None:
            self._text_service = TextPreparationService(self.context)
        return self._text_service

    @property
    def validation_service(self) -> SSMLValidationService:
        if self._validation_service is None:
            self._validation_service = SSMLValidationService(self.context)
        return self._validation_service

    @property
    def audio_service(self) -> AudioGenerationService:
        if self._audio_service is None:
            self._audio_service = AudioGenerationService(self.context)
        return self._audio_service

    def find_book_file(self, book_dir: str = "book") -> Path:
        """
        Ищет файл книги в указанной папке.
        """
        book_path = Path(book_dir)

        if not book_path.exists():
            raise FileNotFoundError(f"Папка '{book_dir}' не существует")

        # Поддерживаемые форматы
        formats = self.deps.get_available_formats()

        book_files = []
        for ext in formats:
            book_files.extend(book_path.glob(f"*.{ext}"))
            book_files.extend(book_path.glob(f"*.{ext.upper()}"))

        book_files = [f for f in book_files if f.is_file()]

        if len(book_files) == 0:
            raise FileNotFoundError(
                f"В папке '{book_dir}' не найдено файлов. "
                f"Поддерживаемые форматы: {', '.join(formats)}"
            )

        if len(book_files) > 1:
            files_list = "\n   ".join(f.name for f in book_files)
            raise ValueError(
                f"В папке '{book_dir}' найдено несколько файлов:\n   {files_list}\n"
                "Оставьте только один файл или укажите его явно через -i"
            )

        return book_files[0]

    def find_book_config(
        self, book_name: str, config_dir: str = "book_specific"
    ) -> Path:
        """
        Ищет конфиг для конкретной книги.
        """
        config_path = Path(config_dir)

        # Сначала ищем специфичный конфиг
        specific = config_path / f"{book_name}.json"
        if specific.exists():
            return specific

        # Потом общий
        default = config_path / "book_config.json"
        if default.exists():
            return default

        return None

    def create_context(
        self,
        input_path: Path,
        output_path: Path = None,
        options: ProcessingOptions = None,
    ) -> ExecutionContext:
        """
        Создаёт контекст выполнения.
        """
        # Определяем выходной путь
        if output_path is None:
            audio_dir = Path("audio")
            audio_dir.mkdir(exist_ok=True)

            ext = ".m4a"  # По умолчанию AAC
            if options:
                if options.output_format == AudioFormat.MP3:
                    ext = ".mp3"
                elif options.output_format == AudioFormat.WAV:
                    ext = ".wav"

            output_path = audio_dir / f"{input_path.stem}{ext}"

        # Создаём контекст
        context = ExecutionContext(
            input_path=input_path,
            output_path=output_path,
            options=options or ProcessingOptions(),
            debug=self.debug,
            deps=self.deps,
            config=self.config,
        )

        # Загружаем конфиг книги если есть
        book_config_path = self.find_book_config(input_path.stem)
        if book_config_path:
            self.config.load_book_config(config_path=book_config_path)
            context.book_config = self.config.book_config
            self.debug.log(
                f"📖 Загружен конфиг книги: {book_config_path}", level="INFO"
            )

        # Определяем режим выполнения
        if self.debug.should_log("VERBOSE"):
            context.options.mode = ProcessingMode.SEQUENTIAL
            self.debug.log(
                "🐛 Режим ОТЛАДКИ: последовательное выполнение", level="INFO"
            )

        # ОТЛАДКА
        self.debug.log(f"Options genre: {context.options.genre}", level='INFO')
        self.debug.log(f"Config genre: {context.config.processing_config.default_genre if context.config else 'no config'}", level='INFO')
        self.debug.log(f"Effective genre: {context.get_effective_genre()}", level='INFO')

        return context

    @debug_trace
    @debug_timing
    def build_pipeline(self) -> Pipeline:
        """
        Строит конвейер обработки.
        """
        pipeline = Pipeline(self.context)

        # Этап 1: Подготовка текста и SSML
        stage1 = PipelineStage(
            name="Подготовка SSML",
            func=lambda ctx: self._run_text_preparation(ctx),
            enabled=True,
            can_parallelize=False,
        )
        pipeline.add_stage(stage1)

        # Этап 2: Валидация SSML (опционально)
        if not self.context.options.skip_validation:
            stage2 = PipelineStage(
                name="Валидация SSML",
                func=lambda ctx: self._run_validation(ctx),
                enabled=True,
                can_parallelize=False,
            )
            pipeline.add_stage(stage2)

        # Этап 3: Генерация аудио
        stage3 = PipelineStage(
            name="Генерация аудио",
            func=lambda ctx: self._run_audio_generation(ctx),
            enabled=True,
            can_parallelize=False,  # Внутри сервиса есть свой параллелизм
        )
        pipeline.add_stage(stage3)

        return pipeline

    def _run_text_preparation(self, context: ExecutionContext) -> dict:
        """
        Выполняет подготовку текста.
        """
        self.debug.log("\n" + "─" * 60, level="INFO")
        self.debug.log("📝 ЭТАП 1: ПОДГОТОВКА SSML", level="INFO")
        self.debug.log("─" * 60, level="INFO")

        # Подготавливаем текст
        ssml_text = self.text_service.process()

        # Разбиваем на чанки
        chunks = self.text_service.prepare_chunks()

        return {"ssml_text": ssml_text, "chunks": chunks, "chunks_count": len(chunks)}

    def _run_validation(self, context: ExecutionContext) -> dict:
        """
        Выполняет валидацию SSML.
        """
        self.debug.log("\n" + "─" * 60, level="INFO")
        self.debug.log("🔍 ЭТАП 2: ВАЛИДАЦИЯ SSML", level="INFO")
        self.debug.log("─" * 60, level="INFO")

        is_valid = self.validation_service.validate(auto_fix=True)

        result = context.validation_result

        if not is_valid:
            self.debug.log_warning(
                "SSML содержит ошибки, но генерация будет продолжена"
            )

        return result

    def _run_audio_generation(self, context: ExecutionContext) -> dict:
        """
        Выполняет генерацию аудио.
        """
        self.debug.log("\n" + "─" * 60, level="INFO")
        self.debug.log("🎙️ ЭТАП 3: ГЕНЕРАЦИЯ АУДИО", level="INFO")
        self.debug.log("─" * 60, level="INFO")

        # Получаем чанки из предыдущего этапа
        chunks_data = context.ssml_chunks
        chunks = [
            SSMLChunk(
                content=c["content"],
                index=c["index"],
                word_count=c.get("word_count", 0),
            )
            for c in chunks_data
        ]

        self.debug.log(f"Чанков для озвучивания: {len(chunks)}", level="INFO")
        self.debug.log(f"Голос: {context.get_effective_voice()}", level="INFO")
        self.debug.log(f"Потоков: {context.options.workers}", level="INFO")

        # Генерируем аудио
        final_path = self.audio_service.generate_full(chunks)

        return {"final_path": final_path, "audio_book": self.audio_service.audio_book}

    def run(
        self,
        input_path: Path = None,
        output_path: Path = None,
        options: ProcessingOptions = None,
    ) -> bool:
        """
        Запускает полный процесс создания аудиокниги.
        """
        self.start_time = time.time()

        try:
            # 1. Находим файл книги если не указан
            if input_path is None:
                input_path = self.find_book_file()
                self.debug.log(f"📖 Найден файл: {input_path.name}", level="INFO")
            elif not input_path.exists():
                raise FileNotFoundError(f"Файл не найден: {input_path}")

            # 2. Создаём контекст
            self.context = self.create_context(input_path, output_path, options)

            # 3. Выводим информацию о книге
            self._print_book_info()

            # 4. Строим и запускаем конвейер
            pipeline = self.build_pipeline()
            self.context = pipeline.run()

            # 5. Проверяем результат
            if self.context.has_errors():
                self.debug.log_error("Процесс завершился с ошибками")
                return False

            # 6. Выводим итоговую информацию
            self._print_final_summary()

            return True

        except AudiobookError as e:
            self.debug.log_error(f"Ошибка аудиокниги: {e}", e, show_traceback=True)
            return False
        except Exception as e:
            self.debug.log_error("Неожиданная ошибка", e, show_traceback=True)
            return False
        finally:
            self.end_time = time.time()

    def _print_book_info(self):
        """Выводит информацию о книге"""
        print("\n" + "="*60)
        print("📖 ИНФОРМАЦИЯ О КНИГЕ")
        print("="*60)
        print(f"   Файл: {self.context.input_path.name}")
        print(f"   Жанр: {self.context.get_effective_genre()}")
        print(f"   Голос: {self.context.get_effective_voice()}")

        # Формат
        if self.context.options.output_formats:
            formats = ", ".join(f.value.upper() for f in self.context.options.output_formats)
            print(f"   Форматы: {formats}")
        else:
            print(f"   Формат: {self.context.options.output_format.value.upper()}")

        print(f"   Режим: {self.context.options.mode.value}")
        print(f"   Потоков: {self.context.get_effective_workers()}")

        # Дополнительная информация
        if self.context.book_config:
            print(f"   📖 Конфиг книги: {self.context.book_config.path.name}")

        print("="*60 + "\n")

    def _print_final_summary(self):
        """Выводит итоговую сводку"""
        duration = 0
        if self.end_time and self.start_time:
            duration = self.end_time - self.start_time

        print("\n" + "=" * 60)
        print("✅ АУДИОКНИГА ГОТОВА!")
        print("=" * 60)

        if self.context and self.context.final_output:
            print(f"   📁 Файл: {self.context.final_output}")

            if self.context.final_output.exists():
                size_mb = self.context.final_output.stat().st_size / (1024 * 1024)
                print(f"   💾 Размер: {size_mb:.2f} MB")

        # Информация из аудио сервиса
        if self._audio_service and self._audio_service.audio_book:
            audio_book = self._audio_service.audio_book
            if audio_book.final_duration_seconds > 0:
                print(f"   ⏱️  Длительность: {audio_book.total_duration_formatted}")

        # Время выполнения
        if duration > 0:
            minutes = int(duration // 60)
            seconds = int(duration % 60)
            print(f"   ⚙️  Время обработки: {minutes} мин {seconds} сек")

        # Статистика
        if self.context and self.context.stats:
            print("\n📊 СТАТИСТИКА:")
            for key, value in self.context.stats.items():
                if isinstance(value, int):
                    print(f"   {key}: {value:,}")
                else:
                    print(f"   {key}: {value}")

        print("=" * 60 + "\n")

def build_options_from_args(args) -> ProcessingOptions:
    """Создаёт ProcessingOptions из аргументов командной строки"""
    options = ProcessingOptions()

    # Основные
    if args.genre:
        options.genre = args.genre
    if args.voice:
        options.voice = args.voice
    if args.chunk_size:
        options.chunk_size = args.chunk_size
    if args.workers:
        options.workers = args.workers

    # Обработка
    options.split_chapters = args.split_chapters
    options.skip_validation = args.skip_validation

    # Вывод
    if args.format:
        options.output_format = AudioFormat(args.format)
    if args.formats:
        options.output_formats = [AudioFormat(f) for f in args.formats]
    if args.bitrate:
        options.bitrate = args.bitrate
    if args.sample_rate:
        options.sample_rate = args.sample_rate
    options.no_convert = args.no_convert

    # Аудио
    options.lowpass = args.lowpass
    if args.lowpass_cutoff:
        options.lowpass_cutoff = args.lowpass_cutoff
    options.no_deesser = args.no_deesser
    if args.deesser_threshold:
        options.deesser_threshold = args.deesser_threshold
    options.no_spectral_nr = args.no_spectral_nr
    if args.spectral_nr_amount:
        options.spectral_nr_amount = args.spectral_nr_amount
    options.room_tone = args.room_tone
    options.no_normalize = args.no_normalize
    if args.target_volume:
        options.target_volume = args.target_volume

    return options


def main():
    """Точка входа"""
    parser = create_parser()
    args = parser.parse_args()

    # Проверяем наличие конфигов, если нет - генерируем
    config_dir = args.config_dir or "config"
    default_config = Path(config_dir) / "default_settings.json"

    if not default_config.exists():
        print("\n⚠️ Конфигурационные файлы не найдены!")
        print("📝 Запуск генератора конфигов...\n")

        from core.config_generator import generate_configs
        generate_configs(config_dir, force=False)
        print("\n✅ Конфиги сгенерированы. Отредактируйте их при необходимости.\n")

    # Инициализируем отладчик
    if args.debug:
        init_debug(level=args.debug, log_to_file=not args.no_debug_log)

    debug = get_debugger()

    try:
        # Создаём опции
        options = build_options_from_args(args)

        # Создаём оркестратор
        orchestrator = AudiobookOrchestrator(debug_level=args.debug)

        # Загружаем конфиг книги если указан явно
        if args.book_config:
            orchestrator.config.load_book_config(config_path=args.book_config)

        # Запускаем
        success = orchestrator.run(
            input_path=args.input, output_path=args.output, options=options
        )

        # Выводим сводку отладки
        if args.debug:
            debug.print_summary()

        sys.exit(0 if success else 1)

    except KeyboardInterrupt:
        print("\n⚠️ Прервано пользователем")
        debug.log_warning("Процесс прерван пользователем")
        sys.exit(1)
    except Exception as e:
        print(f"\n❌ Критическая ошибка: {e}")
        debug.log_error("Критическая ошибка", e, show_traceback=True)
        sys.exit(1)


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
Контекст выполнения - передаётся между всеми этапами обработки
"""

from dataclasses import dataclass, field
from pathlib import Path
from typing import Optional, List, Dict, Any
from enum import Enum


class ProcessingMode(Enum):
    """Режим обработки"""
    SEQUENTIAL = "sequential"
    PARALLEL = "parallel"


class AudioFormat(Enum):
    """Формат выходного аудио"""
    AAC = "aac"
    MP3 = "mp3"
    WAV = "wav"


@dataclass
class ProcessingOptions:
    """Опции обработки"""
    mode: ProcessingMode = ProcessingMode.PARALLEL
    genre: Optional[str] = None
    voice: Optional[str] = None
    sample_rate: Optional[int] = None
    bitrate: Optional[str] = None

# -*- coding: utf-8 -*-

"""
Конвейер обработки - управляет последовательностью этапов
"""

import time
import gc
from typing import Callable, List, Dict, Any
from dataclasses import dataclass, field
from concurrent.futures import ThreadPoolExecutor, as_completed

from core.context import ExecutionContext, ProcessingMode
from core.exceptions import PipelineError
from core.debug_core import get_debugger, debug_trace, debug_timing


@dataclass
class PipelineStage:
    """Этап конвейера"""

    name: str
    func: Callable
    enabled: bool = True
    can_parallelize: bool = False
    depends_on: List[str] = field(default_factory=list)
    metadata: Dict[str, Any] = field(default_factory=dict)


class Pipeline:
    """
    Конвейер обработки.
    Управляет выполнением этапов в правильном порядке.
    """

    def __init__(self, context: ExecutionContext):
        self.context = context
        self.debug = context.debug or get_debugger()
        self.stages: List[PipelineStage] = []
        self.results: Dict[str, Any] = {}
        self.timings: Dict[str, float] = {}

    def add_stage(self, stage: PipelineStage) -> "Pipeline":
        """Добавляет этап в конвейер"""
        self.stages.append(stage)
        self.debug.log(f"➕ Добавлен этап: {stage.name}", level="VERBOSE")
        return self

    def add_stages(self, stages: List[PipelineStage]) -> "Pipeline":
        """Добавляет несколько этапов"""
        for stage in stages:
            self.add_stage(stage)
        return self

    @debug_trace
    @debug_timing
    def _run_stage(self, stage: PipelineStage) -> Any:
        """Выполняет один этап"""
        self.debug.log_stage(stage.name, "start")

        start_time = time.time()

        try:
            result = stage.func(self.context)
            self.results[stage.name] = result
            self.timings[stage.name] = time.time() - start_time

            self.debug.log_stage(stage.name, "end")
            self.debug.log(
                f"   ⏱️ Время: {self.timings[stage.name]:.2f} сек", level="INFO"
            )

            return result
        except Exception as e:
            self.debug.log_stage(stage.name, "error")
            self.context.add_error(f"Ошибка на этапе {stage.name}: {e}")
            raise PipelineError(
                f"Этап {stage.name} завершился с ошибкой", {"stage": stage.name}
            ) from e

    @debug_trace
    def run(self) -> ExecutionContext:
        """
        Запускает конвейер.
        В режиме SEQUENTIAL - этапы выполняются строго последовательно.
        В режиме PARALLEL - этапы, которые можно распараллелить, выполняются параллельно.
        """
        self.debug.log_stage("ЗАПУСК КОНВЕЙЕРА", "start")
        self.debug.log(f"   Режим: {self.context.options.mode.value}", level="INFO")
        self.debug.log(f"   Этапов: {len(self.stages)}", level="INFO")

        if self.context.options.mode == ProcessingMode.SEQUENTIAL:
            return self._run_sequential()
        else:
            return self._run_parallel()

    def _run_sequential(self) -> ExecutionContext:
        """Последовательное выполнение этапов"""
        self.debug.log("🔄 Последовательное выполнение", level="INFO")

        for i, stage in enumerate(self.stages, 1):
            if not stage.enabled:
                self.debug.log(f"⏭️ Пропущен этап: {stage.name}", level="INFO")
                continue

            self.debug.log(
                f"\n--- Этап {i}/{len(self.stages)}: {stage.name} ---", level="INFO"
            )

            try:
                self._run_stage(stage)
            except PipelineError:
                # Останавливаем конвейер при ошибке
                self.debug.log_error(f"❌ Конвейер остановлен на этапе: {stage.name}")
                break

            # Принудительная сборка мусора после тяжёлых этапов
            if "tts" in stage.name.lower() or "audio" in stage.name.lower():
                gc.collect()
                self.debug.log("🧹 Сборка мусора выполнена", level="VERBOSE")

        self._print_summary()
        return self.context

    def _run_parallel(self) -> ExecutionContext:
        """Параллельное выполнение этапов (где возможно)"""
        self.debug.log("⚡ Параллельное выполнение", level="INFO")

        # Группируем этапы: последовательные и параллельные группы
        groups = self._group_stages()

        for group_idx, group in enumerate(groups, 1):
            if len(group) == 1:
                # Один этап - выполняем последовательно
                stage = group[0]
                if stage.enabled:
                    self.debug.log(
                        f"\n--- Этап {group_idx}/{len(groups)}: {stage.name} ---",
                        level="INFO",
                    )
                    try:
                        self._run_stage(stage)
                    except PipelineError:
                        break
            else:
                # Несколько этапов - выполняем параллельно
                self.debug.log(
                    f"\n--- Параллельная группа {group_idx}/{len(groups)}: "
                    + f"{', '.join(s.name for s in group)} ---",
                    level="INFO",
                )

                with ThreadPoolExecutor(max_workers=len(group)) as executor:
                    futures = {}
                    for stage in group:
                        if stage.enabled:
                            futures[executor.submit(self._run_stage, stage)] = stage

                    for future in as_completed(futures):
                        stage = futures[future]
                        try:
                            future.result()
                        except PipelineError as e:
                            self.debug.log_error(
                                f"Ошибка в параллельном этапе {stage.name}: {e}"
                            )
                            # Отменяем остальные
                            for f in futures:
                                f.cancel()
                            break

                gc.collect()

        self._print_summary()
        return self.context

    def _group_stages(self) -> List[List[PipelineStage]]:
        """Группирует этапы для параллельного выполнения"""
        groups = []
        current_group = []

        for stage in self.stages:
            if stage.can_parallelize:
                current_group.append(stage)
            else:
                if current_group:
                    groups.append(current_group)
                    current_group = []
                groups.append([stage])

        if current_group:
            groups.append(current_group)

        return groups

    def _print_summary(self):
        """Выводит сводку выполнения"""
        self.debug.log("\n" + "=" * 60, level="INFO")
        self.debug.log("📊 СВОДКА ВЫПОЛНЕНИЯ КОНВЕЙЕРА", level="INFO")
        self.debug.log("=" * 60, level="INFO")

        total_time = sum(self.timings.values())
        self.debug.log(f"⏱️ Общее время: {total_time:.2f} сек", level="INFO")

        for stage_name, duration in self.timings.items():
            percentage = (duration / total_time * 100) if total_time > 0 else 0
            self.debug.log(
                f"   {stage_name}: {duration:.2f} сек ({percentage:.1f}%)", level="INFO"
            )

        if self.context.errors:
            self.debug.log(f"\n❌ Ошибок: {len(self.context.errors)}", level="INFO")
        if self.context.warnings:
            self.debug.log(
                f"⚠️ Предупреждений: {len(self.context.warnings)}", level="INFO"
            )

        self.debug.log_stage("ЗАВЕРШЕНИЕ КОНВЕЙЕРА", "end")

#!/usr/bin/env python3

Сервис подготовки текста

    @debug_trace
    @debug_timing

        """
        Основной метод обработки.
        Готовит финальный текст со ВСЕМИ трансформациями,
        но НЕ строит полный SSML.
        """
        self.debug.log_stage("Подготовка текста", "start")

        # 1. Извлечение
        raw_text = self._extract_text()
        self.context.raw_text = raw_text

        book = Book.from_file(self.context.input_path, raw_text)
        self.context.metadata['book'] = book.to_dict()

        # 2. Очистка
        cleaned_text, paragraphs = self._clean_text(raw_text)
        self.context.cleaned_text = cleaned_text
        self.context.paragraphs = paragraphs

        # Сохраняем параграфы для SSML
        self.final_paragraphs = paragraphs

        if self.context.options.split_chapters:
            chapters = self._split_into_chapters(cleaned_text)
            self.context.chapters = chapters

        # 3. Применяем ВСЕ трансформации к тексту
        accented_text = self._apply_accents(cleaned_text)
        text_with_numbers = self._convert_numbers(accented_text)
        fixed_text = self._apply_manual_fixes(text_with_numbers)
        expanded_text = self._expand_abbreviations(fixed_text)

        # 4. Сохраняем финальный обработанный текст (с тегами!)
        self.final_processed_text = expanded_text
        self.context.stats['final_chars'] = len(expanded_text)

        self.debug.log_stage("Подготовка текста", "end")
        self.debug.log(f"✅ Финальный текст готов: {len(expanded_text)} символов", level='INFO')

        return expanded_text

    @debug_trace
    def _extract_text(self) -> str:
        self.debug.log(f"Извлечение текста из: {self.context.input_path}", level='INFO')
        try:
            text = self.file_utils.extract_text(self.context.input_path)
            if not text:
                raise TextExtractionError("Текст не найден в файле")
            self.debug.log(f"Извлечено {len(text):,} символов", level='INFO')
            self.context.stats['raw_chars'] = len(text)
            return text
        except Exception as e:
            raise TextExtractionError(f"Ошибка извлечения текста: {e}") from e

    @debug_trace
    def _clean_text(self, text: str) -> Tuple[str, List[str]]:
        self.debug.log("Очистка текста...", level='INFO')
        cleaned, paragraphs = self.cleaner.clean(text)
        self.context.stats['cleaned_chars'] = len(cleaned)
        self.context.stats['paragraphs'] = len(paragraphs)
        return cleaned, paragraphs

    @debug_trace
    def _split_into_chapters(self, text: str) -> List[dict]:
        self.debug.log("Разбиение на главы...", level='INFO')
        chapters_info = self.text_utils.split_into_chapters(text)
        chapters = []
        for i, (title, content) in enumerate(chapters_info):
            chapters.append({'title': title, 'content': content, 'index': i})
        self.context.stats['chapters'] = len(chapters)
        return chapters

    @debug_trace
    def _apply_accents(self, text: str) -> str:
        self.debug.log("Расстановка ударений...", level='INFO')
        try:
            accented = self.accent_engine.apply(text)
            self.context.stats['accents_applied'] = accented.count('+')
            return accented
        except Exception as e:
            raise AccentError(f"Ошибка расстановки ударений: {e}") from e

    @debug_trace
    def _convert_numbers(self, text: str) -> str:
        self.debug.log("Преобразование чисел...", level='INFO')
        return self.numbers_converter.convert(text)

    @debug_trace
    def _apply_manual_fixes(self, text: str) -> str:
        fixes = self.config.get_merged_fixes()
        if not fixes:
            return text
        self.debug.log(f"Применение {len(fixes)} ручных исправлений...", level='VERBOSE')
        for wrong, correct in fixes.items():
            text = text.replace(wrong, correct)
        return text

    @debug_trace
    def _expand_abbreviations(self, text: str) -> str:
        abbreviations = self.config.get_merged_abbreviations()
        if not abbreviations:
            return text
        self.debug.log(f"Расшифровка {len(abbreviations)} аббревиатур...", level='VERBOSE')
        for abbr, full in abbreviations.items():
            text = re.sub(r'\b' + re.escape(abbr) + r'\b', full, text)
        return text

    def _strip_tags(self, text: str) -> str:
        """Удаляет все XML/SSML теги, оставляя только чистый текст."""
        return re.sub(r'<[^>]+>', '', text)

    def _find_original_chunk(self, clean_chunk: str, full_text: str) -> str:
        """
        Находит оригинальный фрагмент в полном тексте (с тегами),
        соответствующий чистому чанку.
        """
        # Упрощённый подход: ищем первое вхождение чистого текста в полном тексте
        # Это работает, потому что чистый текст — это тот же текст, но без тегов
        clean_chunk_escaped = re.escape(clean_chunk)
        pattern = re.compile(clean_chunk_escaped)
        
        match = pattern.search(full_text)
        if match:
            return match.group(0)
        
        # Если не нашли (например, из-за пробелов), возвращаем чистый чанк
        return clean_chunk

    @debug_trace
    def _build_ssml_for_chunk(self, chunk_text: str) -> str:
        """Строит SSML для одного чанка."""
        try:
            ssml = self.ssml_builder.build(chunk_text, [chunk_text])
            return ssml
        except Exception as e:
            raise SSMLBuildError(f"Ошибка построения SSML: {e}") from e

    @debug_trace
    def prepare_chunks(self, chunk_size: int = 400) -> List[SSMLChunk]:
        """
        Разбивает текст на чанки:
        1. Нарезает ЧИСТЫЙ текст (без тегов) по предложениям.
        2. Для каждого чанка находит соответствующий фрагмент в ОРИГИНАЛЬНОМ тексте (с тегами).
        3. Строит SSML из оригинального фрагмента.
        """
        if not self.final_processed_text:
            raise ValueError("Сначала вызови process()")

        # 1. Получаем чистый текст (без тегов) для нарезки
        clean_text = self._strip_tags(self.final_processed_text)

        # 2. Разбиваем чистый текст на предложения
        sentences = self.text_utils.split_sentences(clean_text)

        if not sentences:
            self.debug.log_warning("Не удалось разбить текст на предложения, создаю один чанк")
            sentences = [clean_text]

        self.debug.log(f"Текст разбит на {len(sentences)} предложений", level='INFO')

        # 3. Группируем предложения в чанки (по чистому тексту)
        clean_chunks = []
        current_batch = []
        current_len = 0

        for sent in sentences:
            sent = sent.strip()
            if not sent:
                continue

            if current_len + len(sent) > chunk_size and current_batch:
                clean_chunks.append(" ".join(current_batch))
                current_batch = []
                current_len = 0

            current_batch.append(sent)
            current_len += len(sent)

        if current_batch:
            clean_chunks.append(" ".join(current_batch))

        self.debug.log(f"Предложения сгруппированы в {len(clean_chunks)} чанков", level='INFO')

        # 4. Для каждого чистого чанка находим оригинальный текст (с тегами) и строим SSML
        ssml_chunks = []
        full_text = self.final_processed_text
        
        for i, clean_chunk in enumerate(clean_chunks):
            # Находим оригинальный фрагмент
            original_chunk = self._find_original_chunk(clean_chunk, full_text)
            
            # Строим SSML
            ssml_content = self._build_ssml_for_chunk(original_chunk)
            
            # Сохраняем для отладки
            chunk_path = Path("logs") / f"debug_chunk_{i:04d}.txt"
            with open(chunk_path, 'w', encoding='utf-8') as f:
                f.write(ssml_content)
            
            chunk = SSMLChunk(content=ssml_content, index=i)
            ssml_chunks.append(chunk)
            self.debug.log(f"Чанк {i} сохранён: {chunk_path}", level='VERBOSE')

        # Обновляем контекст
        self.context.ssml_chunks = [
            {'index': c.index, 'content': c.content} for c in ssml_chunks
        ]

        self.debug.log(f"✅ Создано {len(ssml_chunks)} валидных чанков с SSML-разметкой", level='INFO')
        return ssml_chunks

        """
        Генерирует аудио из списка SSML чанков.
        Поддерживает параллельную генерацию.
        """
        self.debug.log_stage("Генерация аудио", "start")
        self.debug.log(f"Чанков для генерации: {len(ssml_chunks)}", level='INFO')
        
        audio_chunks = []
        self._temp_dir = Path(tempfile.mkdtemp(prefix="audiobook_"))
        self.debug.log(f"📁 Временная директория: {self._temp_dir}", level='VERBOSE')
        
        # Подготавливаем задания
        jobs = []
        for chunk in ssml_chunks:
            output_path = self._temp_dir / f"chunk_{chunk.index:04d}.wav"
            jobs.append((chunk, output_path))
        
        workers = self.context.options.workers
        
        if self.context.is_debug_mode():
            # Последовательная генерация в режиме отладки
            audio_chunks = self._generate_sequential(jobs)
        else:
            # Параллельная генерация
            audio_chunks = self._generate_parallel(jobs, workers)

        audio_chunks = self._post_process_chunks(audio_chunks)
        
        self.audio_book.chunks = audio_chunks
        self.context.audio_files = [c.processed_path or c.wav_path for c in audio_chunks if c.is_ready]
        
        self.debug.log_stage("Генерация аудио", "end")
        
        return audio_chunks
    
    def _generate_sequential(self, jobs: List[Tuple]) -> List[AudioChunk]:
        """Последовательная генерация чанков"""
        audio_chunks = []
        
        for i, (chunk, output_path) in enumerate(jobs):
            self.debug.log(f"Генерация чанка {i+1}/{len(jobs)}...", level='INFO')
            
            audio_chunk = AudioChunk(
                ssml_content=chunk.content,
                index=chunk.index,
                chapter_index=chunk.chapter_index,
                status=AudioChunkStatus.GENERATING
            )
            
            try:
                success = self.tts_engine.generate(chunk.content, output_path)
                if success:
                    audio_chunk.status = AudioChunkStatus.COMPLETED
                    audio_chunk.wav_path = output_path
                    
                    # Получаем информацию о файле
                    info = self.audio_utils.get_wav_info(output_path)
                    if info:
                        audio_chunk.duration_seconds = info[2]
                        audio_chunk.sample_rate = info[0]
                        audio_chunk.file_size = output_path.stat().st_size
                else:
                    audio_chunk.status = AudioChunkStatus.FAILED
                    audio_chunk.error = "Ошибка генерации"
                    
            except Exception as e:
                audio_chunk.status = AudioChunkStatus.FAILED
                audio_chunk.error = str(e)
                self.debug.log_error(f"Ошибка чанка {i}", e)
            
            audio_chunks.append(audio_chunk)
            
            # Сборка мусора после каждого чанка в режиме отладки
            gc.collect()
        
        return audio_chunks
    
    def _generate_parallel(self, jobs: List[Tuple], workers: int) -> List[AudioChunk]:
        """Параллельная генерация чанков"""
        audio_chunks = []
        futures = {}
        
        with ThreadPoolExecutor(max_workers=workers) as executor:
            for chunk, output_path in jobs:
                audio_chunk = AudioChunk(
                    ssml_content=chunk.content,
                    index=chunk.index,
                    chapter_index=chunk.chapter_index,
                    status=AudioChunkStatus.PENDING
                )
                audio_chunks.append(audio_chunk)
                
                future = executor.submit(
                    self._generate_single_chunk,
                    chunk.content,
                    output_path,
                    audio_chunk
                )
                futures[future] = audio_chunk
            
            completed = 0
            for future in as_completed(futures):
                audio_chunk = futures[future]
                try:
                    updated_chunk = future.result()
                    # Обновляем оригинальный чанк
                    audio_chunk.status = updated_chunk.status
                    audio_chunk.wav_path = updated_chunk.wav_path
                    audio_chunk.duration_seconds = updated_chunk.duration_seconds
                    audio_chunk.error = updated_chunk.error
                    
                    completed += 1
                    self.debug.log(f"Прогресс TTS: {completed}/{len(jobs)}", level='VERBOSE')
                    
                except Exception as e:
                    audio_chunk.status = AudioChunkStatus.FAILED
                    audio_chunk.error = str(e)
                    self.debug.log_error(f"Ошибка в параллельном чанке", e)
            
            # Сборка мусора после параллельной генерации
            gc.collect()
        
        return audio_chunks
    
    def _generate_single_chunk(self, ssml: str, output_path: Path, 
                               audio_chunk: AudioChunk) -> AudioChunk:
        """Генерирует один чанк (для параллельного выполнения)"""
        audio_chunk.status = AudioChunkStatus.GENERATING
        
        try:
            success = self.tts_engine.generate(ssml, output_path)
            if success:
                audio_chunk.status = AudioChunkStatus.COMPLETED
                audio_chunk.wav_path = output_path
                
                info = self.audio_utils.get_wav_info(output_path)
                if info:
                    audio_chunk.duration_seconds = info[2]
                    audio_chunk.sample_rate = info[0]
                    audio_chunk.file_size = output_path.stat().st_size
            else:
                audio_chunk.status = AudioChunkStatus.FAILED
                audio_chunk.error = "Ошибка генерации"
        except Exception as e:
            audio_chunk.status = AudioChunkStatus.FAILED
            audio_chunk.error = str(e)
        
        return audio_chunk
    
    def _post_process_chunks(self, audio_chunks: List[AudioChunk]) -> List[AudioChunk]:
        """Применяет постобработку к чанкам"""
        self.debug.log("Постобработка аудио...", level='INFO')
        
        for chunk in audio_chunks:
            if chunk.status == AudioChunkStatus.COMPLETED and chunk.wav_path:
                chunk.status = AudioChunkStatus.POST_PROCESSING
                
                processed_path = self._temp_dir / f"chunk_{chunk.index:04d}.processed.wav"
                try:
                    result_path = self.audio_processor.process(chunk.wav_path, processed_path)
                    chunk.processed_path = result_path
                    chunk.status = AudioChunkStatus.COMPLETED
                    
                    # Обновляем информацию
                    if result_path.exists():
                        info = self.audio_utils.get_wav_info(result_path)
                        if info:
                            chunk.duration_seconds = info[2]
                            chunk.file_size = result_path.stat().st_size
                            
                except Exception as e:
                    self.debug.log_error(f"Ошибка постобработки чанка {chunk.index}", e)
                    chunk.processed_path = chunk.wav_path
                    chunk.status = AudioChunkStatus.COMPLETED
        
        return audio_chunks
    
    @debug_trace
    def combine_chunks(self, audio_chunks: List[AudioChunk], output_path: Path) -> Optional[Path]:
        """Объединяет чанки с микро-фейдами для бесшовной склейки"""
        ready_chunks = [c for c in audio_chunks if c.is_ready]

        if not ready_chunks:
            self.debug.log_error("Нет готовых чанков для объединения")
            return None

        ready_chunks.sort(key=lambda c: c.index)
        wav_files = [c.processed_path or c.wav_path for c in ready_chunks]

        self.debug.log(f"Объединение {len(wav_files)} чанков с фейдами...", level='INFO')

        temp_combined = self._temp_dir / f"{output_path.stem}.combined.wav"

        try:
            from pydub import AudioSegment

            combined = AudioSegment.empty()
            fade_ms = 10

            for wav_file in wav_files:
                segment = AudioSegment.from_wav(str(wav_file))
                segment = segment.fade_in(fade_ms).fade_out(fade_ms)
                combined += segment

            combined.export(temp_combined, format="wav")

            self.debug.log(f"✅ Чанки объединены с фейдами: {temp_combined}", level='INFO')

            info = self.audio_utils.get_wav_info(temp_combined)
            if info:
                self.audio_book.final_duration_seconds = info[2]

            return temp_combined

        except ImportError:
            self.debug.log_warning("pydub не установлен, использую обычное объединение")
            if self.audio_utils.concat_wav_files(wav_files, temp_combined):
                info = self.audio_utils.get_wav_info(temp_combined)
                if info:
                    self.audio_book.final_duration_seconds = info[2]
                return temp_combined
            return None

    def convert_to_format(self, input_path: Path, output_path: Path,

        return None
    
    @debug_trace
    def generate_full(self, ssml_chunks: List[SSMLChunk]) -> Optional[Path]:
        """
        Полный цикл: генерация -> постобработка -> объединение -> конвертация.
        """
        # 1. Генерация
        audio_chunks = self.generate_from_ssml(ssml_chunks)
        
        # Проверяем, что есть готовые чанки
        ready_count = sum(1 for c in audio_chunks if c.is_ready)
        if ready_count == 0:
            raise TTSGenerationError("Ни один чанк не был сгенерирован")
        
        self.debug.log(f"Сгенерировано {ready_count}/{len(audio_chunks)} чанков", level='INFO')
        
        # 2. Определяем выходной путь
        output_path = self.context.output_path
        if not output_path:
            output_path = Path("audio") / f"{self.context.input_path.stem}.m4a"
        output_path.parent.mkdir(parents=True, exist_ok=True)
        
        # 3. Объединение (если больше одного чанка)
        if len(audio_chunks) > 1:
            combined = self.combine_chunks(audio_chunks, output_path)
        else:
            combined = audio_chunks[0].processed_path or audio_chunks[0].wav_path
        
        if not combined:
            raise AudioProcessingError("Не удалось объединить чанки")
        
        # 4. Конвертация
        if self.context.options.no_convert:
            final_path = output_path.with_suffix('.wav')
            shutil.copy2(str(combined), str(final_path))
            self.context.final_output = final_path
        else:
            audio_format = self.context.options.output_format
            
            if self.context.options.output_formats:
                final_path = None
                for fmt in self.context.options.output_formats:
                    fmt_path = self.convert_to_format(combined, output_path, fmt)
                    if fmt_path and not final_path:
                        final_path = fmt_path
            else:
                final_path = self.convert_to_format(combined, output_path, audio_format)

        # 5. Сохранение метаданных
        if final_path and final_path.exists():
            self._save_metadata(final_path)

        # 6. Очистка временных файлов
        if not self.context.is_debug_mode():
            self._cleanup_all_temp_files(audio_chunks, combined)
        else:
            self.debug.log(f"🐛 Режим отладки: временные файлы сохранены в {self._temp_dir}", level='INFO')
        
        return final_path

        if self._temp_dir and self._temp_dir.exists():

    def _clean_ssml(self, text: str) -> str:
        import html
        import re

        # 1. Разекранируем (чтобы &lt; стал <)
        text = html.unescape(text)

        # 2. Если чанк все же пришел битым (например, "азывались</prosody>")
        # Мы удаляем все теги, у которых нет пары внутри этого чанка
        if text.count('<') != text.count('>'):
            # Если скобки не парные - вычищаем их вообще, оставляем голый текст
            text = re.sub(r'<[^>]+>', '', text)
            text = text.replace('<', '').replace('>', '')

        # 3. Гарантируем одну обертку <speak>
        text = text.replace('<speak>', '').replace('</speak>', '').strip()

        # 4. Удаляем мусорные куски типа 'rate="90%"', которые остались без тега
        text = re.sub(r'[a-z]+="[^"]*"', '', text)

        return f"<speak>{text}</speak>"

    @debug_trace

                aggressive = re.sub(r'</?prosody[^>]*>', '', original_text)

    @debug_trace
    def generate_chunks(
        self, chunks: List[Tuple[str, Path]], workers: int = 4

        results = []

        from concurrent.futures import ThreadPoolExecutor, as_completed

        with ThreadPoolExecutor(max_workers=workers) as executor:
            futures = {}
            for ssml_text, output_path in chunks:
                future = executor.submit(self._generate_safe, ssml_text, output_path)
                futures[future] = output_path

            for future in as_completed(futures):
                output_path = futures[future]
                try:
                    success, error = future.result()
                    results.append((output_path, success, error))
                except Exception as e:
                    results.append((output_path, False, str(e)))

        return results

    def _generate_safe(
        self, ssml_text: str, output_path: Path
    ) -> Tuple[bool, Optional[str]]:
        try:
            success = self.generate(ssml_text, output_path)
            return success, None
        except Exception as e:
            return False, str(e)

    def list_voices(self) -> List[str]:
        return self.AVAILABLE_VOICES

# -*- coding: utf-8 -*-

"""
Движок обработки аудио
"""

from pathlib import Path
from typing import List
from dataclasses import dataclass

from core.debug_core import get_debugger, debug_trace

@dataclass
class ProcessingOptions:
    """Опции обработки аудио"""

    enable_deesser: bool = True
    deesser_threshold: float = -25.0
    deesser_center_freq: int = 6000
    deesser_q: float = 1.0

    enable_spectral_nr: bool = True
    spectral_nr_amount: float = 0.5

    enable_room_tone: bool = False
    room_tone_level: float = -70.0

    enable_lowpass: bool = False
    lowpass_cutoff: int = 4000

    enable_normalization: bool = True
    target_volume: float = -16.0


class AudioProcessor:
    """
    Движок постобработки аудио.
    Применяет de-esser, шумоподавление, нормализацию и т.д.
    """

    def __init__(
        self, options: ProcessingOptions = None, config: AudioProcessingConfig = None
    ):
        self.debug = get_debugger()
        self.audio_utils = AudioUtils()

        if options:
            self.options = options
        elif config:
            self.options = ProcessingOptions(
                enable_deesser=config.enable_deesser,
                deesser_threshold=config.deesser_threshold,
                deesser_center_freq=config.deesser_center_freq,
                deesser_q=config.deesser_q,
                enable_spectral_nr=config.enable_spectral_nr,
                spectral_nr_amount=config.spectral_nr_amount,
                enable_room_tone=config.enable_room_tone,
                room_tone_level=config.room_tone_level,
                enable_lowpass=config.enable_lowpass_filter,
                lowpass_cutoff=config.lowpass_cutoff_freq,
                enable_normalization=config.enable_normalization,
                target_volume=config.target_volume,
            )
        else:
            self.options = ProcessingOptions()

    @debug_trace
    def process(self, input_path: Path, output_path: Path = None) -> Path:
        """
        Применяет все этапы постобработки к аудиофайлу.
        """
        if output_path is None:
            output_path = input_path.with_suffix(".processed.wav")

        current_file = input_path
        temp_files = []

        # De-esser
        if self.options.enable_deesser:
            deessed = current_file.with_suffix(".deessed.wav")
            if self.audio_utils.apply_deesser(
                current_file,
                deessed,
                self.options.deesser_threshold,
                self.options.deesser_center_freq,
                self.options.deesser_q,
            ):
                current_file = deessed
                temp_files.append(deessed)

        # Шумоподавление
        if self.options.enable_spectral_nr:
            nr_file = current_file.with_suffix(".nr.wav")
            if self.audio_utils.apply_spectral_nr(
                current_file, nr_file, self.options.spectral_nr_amount
            ):
                current_file = nr_file
                temp_files.append(nr_file)

        # Room tone
        if self.options.enable_room_tone:
            room_file = current_file.with_suffix(".room.wav")
            if self.audio_utils.add_room_tone(
                current_file, room_file, self.options.room_tone_level
            ):
                current_file = room_file
                temp_files.append(room_file)

        # Low-pass фильтр
        if self.options.enable_lowpass:
            filtered = current_file.with_suffix(".filtered.wav")
            if self.audio_utils.apply_lowpass(
                current_file, filtered, self.options.lowpass_cutoff
            ):
                current_file = filtered
                temp_files.append(filtered)

        # Нормализация
        if self.options.enable_normalization:
            normalized = current_file.with_suffix(".norm.wav")
            if self.audio_utils.normalize_loudness(
                current_file, normalized, self.options.target_volume
            ):
                current_file = normalized
                temp_files.append(normalized)

        # Копируем результат в output_path если нужно
        if current_file != output_path:
            import shutil

            shutil.copy2(str(current_file), str(output_path))

        # Удаляем временные файлы
        for temp in temp_files:
            if temp.exists() and temp != output_path:
                temp.unlink()

        self.debug.log(f"✅ Постобработка завершена: {output_path}", level="INFO")
        return output_path

    @debug_trace
    def process_batch(self, files: List[Path], output_dir: Path = None) -> List[Path]:
        """
        Обрабатывает несколько аудиофайлов.
        """
        results = []

        for i, file in enumerate(files):
            if output_dir:
                output_path = output_dir / f"processed_{i:04d}.wav"
            else:
                output_path = None

            processed = self.process(file, output_path)
            results.append(processed)

            self.debug.log(f"Прогресс обработки: {i + 1}/{len(files)}", level="VERBOSE")

        return results

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
Движок построения SSML разметки (версия с lxml)
"""

import re
import html
from typing import Dict, List, Optional
from lxml import etree

from core.debug_core import get_debugger, debug_trace
from core.config_manager import PauseConfig, ConfigManager
from core.dependency_manager import get_dependency_manager
from utils.text_utils import TextUtils


class SSMLBuilder:
    """Движок построения SSML разметки с использованием lxml для чистого XML."""

    UNSUPPORTED_PROSPY_ATTRS = ['volume', 'contour', 'range', 'duration']

    def __init__(
        self, pause_config: PauseConfig = None, config_manager: ConfigManager = None
    ):
        self.debug = get_debugger()
        self.deps = get_dependency_manager()
        self.text_utils = TextUtils()

        self.pause_config = pause_config or PauseConfig()
        self.config_manager = config_manager

        self.effect_to_ssml = {}
        self.ramp_effects = {}

        if config_manager:
            self.effect_to_ssml = config_manager.EFFECT_TO_SSML
            self.ramp_effects = config_manager.RAMP_EFFECTS

        self.keywords_effects: Dict[str, str] = {}

        self.NSMAP = {None: "http://www.w3.org/2001/10/synthesis"}
        self.SPEAK = f"{{{self.NSMAP[None]}}}speak"
        self.BREAK = f"{{{self.NSMAP[None]}}}break"
        self.PROSODY = f"{{{self.NSMAP[None]}}}prosody"
        self.SAY_AS = f"{{{self.NSMAP[None]}}}say-as"
        self.EMPHASIS = f"{{{self.NSMAP[None]}}}emphasis"

        self.nlp = None
        self._init_syntax_analyzer()

    @debug_trace
    def _init_syntax_analyzer(self):
        if self.deps.is_available("natasha"):
            try:
                from natasha import Segmenter, MorphVocab, NewsEmbedding
                from natasha import NewsMorphTagger, NewsSyntaxParser

                self.segmenter = Segmenter()
                self.morph_vocab = MorphVocab()
                emb = NewsEmbedding()
                self.morph_tagger = NewsMorphTagger(emb)
                self.syntax_parser = NewsSyntaxParser(emb)
                self.nlp = "natasha"
                self.debug.log("✅ Natasha готов", level="INFO")
            except Exception as e:
                self.debug.log_warning(f"Ошибка Natasha: {e}")

    def set_keywords_effects(self, effects: Dict[str, str]):
        self.keywords_effects = effects

    @debug_trace
    def build(self, text: str, paragraphs: List[str] = None) -> str:
        self.debug.log("Построение SSML разметки (lxml)...", level="INFO")

        if paragraphs is None:
            paragraphs = text.split("\n")

        speak_elem = etree.Element(self.SPEAK, nsmap=self.NSMAP)

        for para in paragraphs:
            para = para.strip()
            if not para:
                continue

            self._process_paragraph(para, speak_elem)

            break_elem = etree.SubElement(speak_elem, self.BREAK)
            break_elem.set("time", f"{self.pause_config.paragraph}ms")

        ssml_bytes = etree.tostring(
            speak_elem,
            encoding='utf-8',
            xml_declaration=False,
            pretty_print=False
        )
        ssml_text = ssml_bytes.decode('utf-8')

        # Удаляем пустые теги prosody
        ssml_text = re.sub(r'<prosody\s*></prosody>', '', ssml_text)
        ssml_text = re.sub(r'<prosody\s+></prosody>', '', ssml_text)
        ssml_text = re.sub(r'<prosody\s*>\s*</prosody>', '', ssml_text)

        ssml_text = self._strip_namespaces(ssml_text)
        ssml_text = self._remove_unsupported_attrs(ssml_text)

        # Финальная проверка: не оставляем пустые теги
        ssml_text = re.sub(r'<prosody\s*></prosody>', '', ssml_text)

        self.debug.log("✅ SSML разметка построена", level="INFO")
        return ssml_text

    def _process_paragraph(self, paragraph: str, parent_elem):
        if re.match(r'^\s*[—\-]', paragraph):
            prosody = etree.SubElement(parent_elem, self.PROSODY)
            prosody.set("pitch", "medium")
            prosody.set("rate", "105%")
            self._add_text_with_effects(paragraph, prosody)
        else:
            self._add_text_with_effects(paragraph, parent_elem)

    def _add_text_with_effects(self, text: str, parent_elem):
        if self.keywords_effects:
            text = self._apply_keyword_effects_to_text(text)

        text = self._process_long_words_in_text(text)

        self._add_safe_fragment(text, parent_elem)

    def _add_safe_fragment(self, text: str, parent_elem):
        if not text:
            return

        text = html.unescape(text)

        if '<' in text and '>' in text:
            try:
                wrapped = f"<span>{text}</span>"
                parser = etree.XMLParser(recover=True, remove_blank_text=True)
                fragment = etree.fromstring(wrapped.encode('utf-8'), parser=parser)

                for child in fragment.getchildren():
                    parent_elem.append(child)

                if fragment.tail:
                    self._append_text(parent_elem, fragment.tail)
                elif fragment.text and not list(fragment):
                    self._append_text(parent_elem, fragment.text)

                if len(fragment) > 0 and fragment[-1].tail:
                    self._append_text(parent_elem, fragment[-1].tail)

                return
            except Exception:
                pass

        self._append_text(parent_elem, text)

    def _append_text(self, element, text):
        if len(element) > 0:
            last = element[-1]
            last.tail = (last.tail or "") + text
        else:
            element.text = (element.text or "") + text

    def _apply_keyword_effects_to_text(self, text: str) -> str:
        if not self.keywords_effects:
            return text

        for word, effect in self.keywords_effects.items():
            if effect in self.effect_to_ssml:
                tag_start, tag_name = self.effect_to_ssml[effect]
                pattern = rf'\b({re.escape(word)})\b'
                replacement = rf'<{tag_start}>\1</{tag_name}>'
                text = re.sub(pattern, replacement, text, flags=re.IGNORECASE)

        return text

    def _process_long_words_in_text(self, text: str) -> str:
        """Применяет prosody rate='90%' к длинным словам (>=9 букв)."""
        def repl(match):
            word = match.group(0)
            clean_word = word.replace('+', '')
            # Только если слово длинное и НЕ содержит пробелов
            if len(clean_word) >= 9 and ' ' not in clean_word:
                return f'<prosody rate="90%">{word}</prosody>'
            return word

        protected_pattern = r'<(say-as|prosody)[^>]*>.*?</\1>'

        result_parts = []
        last_end = 0
        for match in re.finditer(protected_pattern, text, re.DOTALL):
            before = text[last_end:match.start()]
            if before:
                parts = TextUtils.split_by_tags(before)
                for i, part in enumerate(parts):
                    if not part.startswith('<'):
                        parts[i] = re.sub(r'\b[А-Яа-яёЁ]+\b', repl, part)
                result_parts.append("".join(parts))
            result_parts.append(match.group(0))
            last_end = match.end()

        if last_end < len(text):
            before = text[last_end:]
            parts = TextUtils.split_by_tags(before)
            for i, part in enumerate(parts):
                if not part.startswith('<'):
                    parts[i] = re.sub(r'\b[А-Яа-яёЁ]+\b', repl, part)
            result_parts.append("".join(parts))

        result = "".join(result_parts)
        # Удаляем пустые теги prosody
        result = re.sub(r'<prosody\s*></prosody>', '', result)
        result = re.sub(r'<prosody\s+></prosody>', '', result)
        return result

    def _strip_namespaces(self, ssml_text: str) -> str:
        ssml_text = re.sub(r'\s+xmlns(:\w+)?="[^"]*"', '', ssml_text)
        ssml_text = re.sub(r'</?ns0:', '</', ssml_text)
        ssml_text = re.sub(r'<ns0:', '<', ssml_text)
        return ssml_text

    def _remove_unsupported_attrs(self, ssml_text: str) -> str:
        for attr in self.UNSUPPORTED_PROSPY_ATTRS:
            ssml_text = re.sub(rf'\s+{attr}="[^"]*"', '', ssml_text)
            ssml_text = re.sub(rf"\s+{attr}='[^']*'", '', ssml_text)
        ssml_text = re.sub(r'<prosody\s*>', '<prosody>', ssml_text)
        ssml_text = re.sub(r'<prosody[^>]*>\s*</prosody>', '', ssml_text)
        return ssml_text


class NumbersToSayAs:
    """Преобразует числа и даты в SSML say-as теги"""

    def __init__(self):
        self.debug = get_debugger()
        self.deps = get_dependency_manager()

    @debug_trace
    def convert(self, text: str) -> str:
        text = self._convert_dates(text)
        text = self._convert_time(text)
        if self.deps.is_available("num2words"):
            text = self._convert_numbers(text)
        return text

    def _convert_dates(self, text: str) -> str:
        def date_repl(match):
            day, month, year = match.groups()
            return f'<say-as interpret-as="date" format="dmy">{day}.{month}.{year}</say-as>'

        parts = TextUtils.split_by_tags(text)
        for i, part in enumerate(parts):
            if not part.startswith("<"):
                parts[i] = re.sub(r"\b(\d{1,2})\.(\d{1,2})\.(\d{4})\b", date_repl, part)
        return "".join(parts)

    def _convert_time(self, text: str) -> str:
        def time_repl(match):
            hour, minute = match.groups()
            return f'<say-as interpret-as="time">{hour}:{minute}</say-as>'

        parts = TextUtils.split_by_tags(text)
        for i, part in enumerate(parts):
            if not part.startswith("<"):
                parts[i] = re.sub(r"\b(\d{1,2}):(\d{2})\b", time_repl, part)
        return "".join(parts)

    def _convert_numbers(self, text: str) -> str:
        if not self.deps.is_available("num2words"):
            return text

        def number_repl(match):
            num_str = match.group(0).replace(" ", "")
            try:
                num = int(num_str)
                if num >= 1000:
                    words = self.deps.num2words.num2words(num, lang="ru")
                    return f'<say-as interpret-as="cardinal">{words}</say-as>'
            except:
                pass
            return match.group(0)

        parts = TextUtils.split_by_tags(text)
        for i, part in enumerate(parts):
            if not part.startswith("<"):
                parts[i] = re.sub(r"\b\d[\d\s]{3,}\b", number_repl, part)
        text = "".join(parts)

        protected_pattern = r'<(say-as|prosody)[^>]*>.*?</\1>'

        result_parts = []
        last_end = 0
        for match in re.finditer(protected_pattern, text, re.DOTALL):
            before = text[last_end:match.start()]
            if before:
                result_parts.append(self._process_long_words_safe(before))
            result_parts.append(match.group(0))
            last_end = match.end()

        if last_end < len(text):
            result_parts.append(self._process_long_words_safe(text[last_end:]))

        return "".join(result_parts)

    def _process_long_words_safe(self, text: str) -> str:
        def repl(match):
            word = match.group(0)
            clean_word = word.replace('+', '')
            if len(clean_word) >= 9 and ' ' not in clean_word:
                return f'<prosody rate="90%">{word}</prosody>'
            return word

        parts = TextUtils.split_by_tags(text)
        for i, part in enumerate(parts):
            if not part.startswith('<'):
                parts[i] = re.sub(r'\b[А-Яа-яёЁ]+\b', repl, part)

        result = "".join(parts)
        result = re.sub(r'<prosody\s*></prosody>', '', result)
        return result
